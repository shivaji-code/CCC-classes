## Question 1: Sort the characters of a given string in ascending order.

```cpp
#include <iostream>
#include <string>
using namespace std;

int main() {
    string s;
    cin >> s;

    for (int i = 0; i < s.length() - 1; i++) {
        for (int j = 0; j < s.length() - i - 1; j++) {
            if (s[j] > s[j + 1]) {
                char temp = s[j];
                s[j] = s[j + 1];
                s[j + 1] = temp;
            }
        }
    }

    cout << s;

    return 0;
}
```

**Input:** `programiz`
**Output:** `agimoprrz`

## Question 2: Find the sum of all numbers embedded in a string.

```cpp
#include <iostream>
#include <string>
using namespace std;

int main() {
    string s;
    cin >> s;

    int sum = 0;
    int num = 0;

    for (int i = 0; i < s.length(); i++) {
        if (s[i] >= '0' && s[i] <= '9') {
            num = num * 10 + (s[i] - '0');
        } else {
            sum = sum + num;
            num = 0;
        }
    }

    sum = sum + num;   // Add the last number if string ends with a digit

    cout << sum;

    return 0;
}
```

**Input:** `abc12de34fg5`
**Output:** `51`

## Question 3: Check whether two given strings are anagrams of each other.

```cpp
#include <iostream>
#include <algorithm>
using namespace std;

int main() {
    string s1, s2;

    cin >> s1 >> s2;

    // Check length first
    if (s1.length() != s2.length()) {
        cout << "Not Anagram";
        return 0;
    }

    // Sort both strings
    sort(s1.begin(), s1.end());
    sort(s2.begin(), s2.end());

    // Compare
    if (s1 == s2)
        cout << "Anagram";
    else
        cout << "Not Anagram";

    return 0;
}
```

**Input:** `listen silent`
**Output:** `Anagram`
